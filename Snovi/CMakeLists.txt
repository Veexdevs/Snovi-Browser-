cmake_minimum_required(VERSION 3.16)
project(Snovi VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT UNIX OR APPLE)
  message(FATAL_ERROR "Esta versão inicial da rede suporta Linux/POSIX; outras plataformas ainda não foram portadas.")
endif()
find_package(OpenSSL REQUIRED)
add_library(snovi_core Network/Url.cpp Network/Http.cpp HTML/Document.cpp Renderer/TextRenderer.cpp AdBlock/AdBlock.cpp Browser/Browser.cpp)
target_include_directories(snovi_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(snovi_core PUBLIC OpenSSL::SSL OpenSSL::Crypto)
target_compile_options(snovi_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(snovi UI/Menu/main.cpp)
target_link_libraries(snovi PRIVATE snovi_core)
enable_testing()
add_executable(snovi_tests tests/tests.cpp)
target_link_libraries(snovi_tests PRIVATE snovi_core)
add_test(NAME core COMMAND snovi_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME network_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_test.py $<TARGET_FILE:snovi>)
endif()
