cmake_minimum_required(VERSION 3.16)
project(Snovi VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
 add_compile_options(/utf-8)
endif()
if(NOT UNIX AND NOT WIN32)
  message(FATAL_ERROR "Plataforma de rede ainda não suportada.")
endif()
find_package(OpenSSL REQUIRED)
add_library(snovi_core Network/Url.cpp Network/Http.cpp HTML/Document.cpp Renderer/TextRenderer.cpp AdBlock/AdBlock.cpp Browser/Browser.cpp)
set_target_properties(snovi_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(snovi_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(snovi_core PUBLIC OpenSSL::SSL OpenSSL::Crypto)
if(MSVC)
 target_compile_options(snovi_core PRIVATE /W4)
else()
 target_compile_options(snovi_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
 target_link_libraries(snovi_core PUBLIC ws2_32 crypt32)
endif()
add_library(snovi_bridge SHARED Platform/Bridge.cpp)
set_target_properties(snovi_bridge PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_link_libraries(snovi_bridge PRIVATE snovi_core)
add_executable(snovi UI/Menu/main.cpp)
target_link_libraries(snovi PRIVATE snovi_core)
enable_testing()
add_executable(snovi_tests tests/tests.cpp)
target_link_libraries(snovi_tests PRIVATE snovi_core)
add_test(NAME core COMMAND snovi_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME desktop_bridge COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/bridge_test.py $<TARGET_FILE:snovi_bridge>)
 add_test(NAME network_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_test.py $<TARGET_FILE:snovi>)
 option(SNOVI_TEST_DESKTOP "Run the optional PySide6 offscreen smoke test" OFF)
 if(SNOVI_TEST_DESKTOP)
  add_test(NAME desktop_gui COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/desktop_test.py $<TARGET_FILE:snovi_bridge>)
  set_tests_properties(desktop_gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 endif()
endif()
